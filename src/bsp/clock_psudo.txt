set a global STATUS variable initial as 0x00 (no fault)
set a global CPU_F variable initiate as 16MHz

if software_reset
    set STATUS to SW_RESET
    jump done




enable TRCENA in the DEMCR register
set DWT->CYCCNT to 0
enable CYCCNTENA in the DWT->CTRL register to start te counter
// now when using the wait we can use the DWT to measure timeouts.

set wait states to 4 WS (5 CPU cycles) in the FLASH_ACR register
check that the new WS value is changed by reading the FLASH_ACR register
set PWREN bit in RCC_APB1ENR1 register to start the pwr clk and enable the change of boost mode
Read RCC_APB1ENR1 to confirm change
set voltage to boost mode by clearing R1MODE bit in PWR_CR5 register(reset value of PWR_CR1 is range 1)
// Now the presets for 170MHz are ready

check pll is off
(if off)
enable HSE
wait for HSE to be stable (max 100ms (100ms is 1 600 000 clk cycles in 16MHz), jump HSE_FAULT if timeout)
// wait logic would be a timestamp named start, the expresion to check is now-start >= timeout(1 600 000 for 100ms 16MHz)
set HSE as PLL input
set PLLM to 6
jump PLL_CONFIG

//if something gets stuck and a wait loop goes over 100ms
HSE_FAULT:
log HSE_FAULT in STATUS variable
disable HSE
set HSI as PLL input
set PLLM to 4

PLL_CONFIG:
set PLLN to 85
set PLLR to 2
Turn on PLL
wait for pll to be stable (check ready flag, max 100ms(100ms is 1 600 000 clk cycles in 16MHz), jump PLL_FAULT if timeout)
Turn on PLLR
Set ahb(HPRE) to divide by 2
set apb1(PPRE1) to divide by 1
set apb2(PPRE2) to divide by 1
read that ahb changed
switch sysclk to PLL
wait for clock switch (check flag, max 100ms(100ms is 8 500 000 clk cycles in 85MHz), jump CLK_SW_FAULT if timeout)
wait for a minimum of 1us(1us is 85 clk cycles in 85MHz)
set ahb(HPRE) to to default value
read that ahb changed
set CPU_F variable to 170MHZ
jump done


CLK_SW_FAULT:
log CLK_SW_FAULT in STATUS variable
switch sysclk to HSI
wait for clock switch (check flag, max 100ms(100ms is 8 500 000 clk cycles in 85MHz), jump RESTART if timeout)
jump DEFAULT_STATE

PLL_FAULT:
log PLL_FAULT in STATUS variable

DEFAULT_STATE:
turn off pll
turn off HSE
set ahb to default value
set apb1/2 to default values
read that ahb changed
set wait states to 0 WS (1 CPU cycles) in the FLASH_ACR register
check that the new WS value is changed by reading the FLASH_ACR register
set voltage to normal mode by setting R1MODE bit in PWR_CR5 register(reset value of PWR_CR1 is range 1)
jump done

RESTART:
set SYSRESETREQ bit in AIRCR register
wait for reset
